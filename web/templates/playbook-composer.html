// State let playbookId = {{if .Playbook}}{{.Playbook.ID}}{{else}}0{{end}}; let currentEditStep = null; let hasChanges = false; // Drag and drop for components function dragStart(event) { event.dataTransfer.setData('text/plain', JSON.stringify({ type: 'component', ref: event.target.dataset.componentRef, name: event.target.dataset.componentName, isRaw: event.target.dataset.isRaw === 'true' })); event.target.classList.add('opacity-50'); } // Drag and drop for existing steps (reordering) function dragStepStart(event) { event.dataTransfer.setData('text/plain', JSON.stringify({ type: 'step', ref: event.target.dataset.componentRef })); event.target.classList.add('opacity-50'); } function dragOver(event) { event.preventDefault(); event.currentTarget.classList.add('border-accent-500', 'bg-accent-500/5'); } function dragLeave(event) { event.currentTarget.classList.remove('border-accent-500', 'bg-accent-500/5'); } function drop(event, phase) { event.preventDefault(); event.currentTarget.classList.remove('border-accent-500', 'bg-accent-500/5'); const data = JSON.parse(event.dataTransfer.getData('text/plain')); const container = document.getElementById('steps-' + phase); if (data.type === 'component') { // Adding new component addStep(container, data.ref, data.name, phase, data.isRaw); } updateEmptyStates(); hasChanges = true; } function addStep(container, ref, name, phase, isRaw) { const step = document.createElement('div'); step.className = 'step-item bg-dark-bg border border-dark-border rounded-lg p-3 flex items-center space-x-3'; step.dataset.componentRef = ref; step.dataset.phase = phase; step.draggable = true; step.ondragstart = dragStepStart; let rawBadge = isRaw ? 'RAW' : ''; step.innerHTML = ` ${rawBadge} ${escapeHtml(ref)} `; container.appendChild(step); } function removeStep(element) { element.remove(); updateEmptyStates(); hasChanges = true; } function editStep(element) { currentEditStep = element; document.getElementById('step-component-ref').value = element.dataset.componentRef; document.getElementById('step-condition').value = element.dataset.condition || ''; document.getElementById('step-edit-modal').classList.remove('hidden'); } function saveStepEdit() { if (currentEditStep) { currentEditStep.dataset.condition = document.getElementById('step-condition').value; } document.getElementById('step-edit-modal').classList.add('hidden'); hasChanges = true; } function updateEmptyStates() { ['pre_deploy', 'deploy', 'post_deploy', 'rollback'].forEach(phase => { const container = document.getElementById('steps-' + phase); const emptyState = container.parentElement.querySelector('.empty-state'); if (container.children.length === 0) { emptyState.style.display = 'block'; } else { emptyState.style.display = 'none'; } }); } function collectSteps() { const steps = []; let order = 1; ['pre_deploy', 'deploy', 'post_deploy', 'rollback'].forEach(phase => { const container = document.getElementById('steps-' + phase); container.querySelectorAll('.step-item').forEach(item => { steps.push({ order: order++, component_ref: item.dataset.componentRef, phase: phase, condition: item.dataset.condition || '', variable_bindings: item.dataset.bindings ? JSON.parse(item.dataset.bindings) : {} }); }); }); return steps; } function validatePlaybook() { const steps = collectSteps(); const resultsDiv = document.getElementById('validation-results'); const contentDiv = document.getElementById('validation-content'); if (steps.length === 0) { contentDiv.innerHTML = '
No steps defined
'; } else { contentDiv.innerHTML = '
Playbook is valid (' + steps.length + ' steps)
'; } resultsDiv.classList.remove('hidden'); } function savePlaybook() { if (!playbookId) { alert('No playbook selected'); return; } const steps = collectSteps(); fetch('/api/v1/recipes/playbooks/' + playbookId + '/steps', { method: 'PUT', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({steps: steps}) }) .then(response => { if (response.ok) { hasChanges = false; alert('Playbook saved successfully'); } else { alert('Failed to save playbook'); } }) .catch(err => { alert('Error saving playbook: ' + err.message); }); } function escapeHtml(text) { const div = document.createElement('div'); div.textContent = text; return div.innerHTML; } // Initialize drop zones document.querySelectorAll('.step-drop-zone').forEach(zone => { zone.addEventListener('dragleave', dragLeave); }); // Update empty states on load updateEmptyStates(); // Warn about unsaved changes window.addEventListener('beforeunload', function(e) { if (hasChanges) { e.preventDefault(); e.returnValue = ''; } }); {{end}}
